# Implemente sua versão modificada de gerar_tempos abaixo e gere uma nova base

def gerar_tempos_uniforme(objetivo, usou_busca, faixa_horaria):
    """Versão modificada: tempo de leitura com distribuição uniforme em vez de log-normal."""
    params_busca = {
        'buscar_sistema': (4.0, 0.55), 'buscar_documento': (4.0, 0.55),
        'buscar_pessoa': (4.0, 0.55), 'ler_comunicado': (2.9, 0.45),
        'acompanhar_noticias': (3.2, 0.50),
    }
    mu_b, sig_b = params_busca.get(objetivo, (3.5, 0.50))
    tempo_busca = np.random.lognormal(mu_b, sig_b)  # mantemos log-normal na busca

    # *** MODIFICAÇÃO: tempo de leitura agora é uniforme ***
    # Ajuste os limites para manter uma média similar
    lim_inf = 30
    lim_sup = 400 if objetivo in {'ler_comunicado', 'acompanhar_noticias'} else 200
    tempo_leitura = np.random.uniform(lim_inf, lim_sup)

    if not usou_busca:
        tempo_busca *= 0.45
    if faixa_horaria == '08-11':
        tempo_leitura *= 0.88
        tempo_busca   *= 1.08
    ruido = np.random.gamma(shape=2.0, scale=8.0)
    tempo_total = tempo_busca + tempo_leitura + ruido
    return tempo_busca, tempo_leitura, tempo_total

# ─── PASSO 2 | Teste KS: aderência à distribuição log-normal ─────────────────
# Extensão: usamos KS (mais preciso que qui-quadrado para distribuições contínuas)
from scipy.stats import kstest, lognorm
import numpy as np

def testar_aderencia_lognormal(serie, nome):
    dados = serie.dropna().values
    dados = dados[dados > 0]
    shape, loc, scale = lognorm.fit(dados, floc=0)
    stat, p = kstest(dados, 'lognorm', args=(shape, loc, scale))
    mu_hat  = np.log(scale)
    sig_hat = shape
    status = 'NÃO rejeita log-normal ✅' if p > 0.05 else 'REJEITA log-normal ❌'
    print(f'  {nome}')
    print(f'    mu_hat={mu_hat:.3f}  sigma_hat={sig_hat:.3f}')
    print(f'    KS stat={stat:.4f}  p-valor={p:.4f}  → {status}')
    print()
    return stat, p

print('Teste Kolmogorov-Smirnov: aderência à distribuição log-normal')

def checar_coerencia(df_in, nome):
    m0 = df_in[df_in['clicou_comunicado']==0]['tempo_leitura_segundos'].median()
    m1 = df_in[df_in['clicou_comunicado']==1]['tempo_leitura_segundos'].median()
    ok = m1 > m0
    status = '✅ direção correta' if ok else '❌ direção INVERTIDA'
    print(f'  {nome}: não_clicou={m0:.0f}s | clicou={m1:.0f}s | diff={m1-m0:+.0f}s  → {status}')
    return m0, m1, ok
