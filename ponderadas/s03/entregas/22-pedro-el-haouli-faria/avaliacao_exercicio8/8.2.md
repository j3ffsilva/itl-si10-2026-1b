# Implemente sua versão modificada de gerar_tempos abaixo e gere uma nova base

def gerar_tempos_uniforme(objetivo, usou_busca, faixa_horaria):
    """Versão modificada: tempo de leitura com distribuição uniforme em vez de log-normal."""
    params_busca = {
        'buscar_sistema': (4.0, 0.55), 'buscar_documento': (4.0, 0.55),
        'buscar_pessoa': (4.0, 0.55), 'ler_comunicado': (2.9, 0.45),
        'acompanhar_noticias': (3.2, 0.50),
    }
    mu_b, sig_b = params_busca.get(objetivo, (3.5, 0.50))
    tempo_busca = np.random.lognormal(mu_b, sig_b)  # mantemos log-normal na busca

    # *** MODIFICAÇÃO: tempo de leitura agora é uniforme ***
    # Ajuste os limites para manter uma média similar
    lim_inf = 30
    lim_sup = 400 if objetivo in {'ler_comunicado', 'acompanhar_noticias'} else 200
    tempo_leitura = np.random.uniform(lim_inf, lim_sup)

    if not usou_busca:
        tempo_busca *= 0.45
    if faixa_horaria == '08-11':
        tempo_leitura *= 0.88
        tempo_busca   *= 1.08
    ruido = np.random.gamma(shape=2.0, scale=8.0)
    tempo_total = tempo_busca + tempo_leitura + ruido
    return tempo_busca, tempo_leitura, tempo_total

# 8.2 — Teste de aderência KS: log-normal vs uniforme
from scipy import stats as st82

tempos_orig = df['tempo_leitura_segundos']
tempos_unif = df_uniforme['tempo_leitura_segundos']

shape_ln, loc_ln, scale_ln = st82.lognorm.fit(tempos_orig, floc=0)

ks_orig = st82.kstest(tempos_orig, 'lognorm', args=(shape_ln, loc_ln, scale_ln))
ks_unif = st82.kstest(tempos_unif, 'lognorm', args=(shape_ln, loc_ln, scale_ln))

print('Teste KS: tempo_leitura vs. log-normal ajustada')
print(f'  Original:  KS={ks_orig.statistic:.4f}  p={ks_orig.pvalue:.4f}  '
      f'{"✅" if ks_orig.pvalue > 0.05 else "⚠️"}')
print(f'  Uniforme:  KS={ks_unif.statistic:.4f}  p={ks_unif.pvalue:.4f}  '
      f'{"✅" if ks_unif.pvalue > 0.05 else "❌"}')
print()
print(f'Descritivas:')
print(f'  Original — μ={tempos_orig.mean():.1f}  med={tempos_orig.median():.1f}  '
      f'σ={tempos_orig.std():.1f}  skew={tempos_orig.skew():.2f}')
print(f'  Uniforme — μ={tempos_unif.mean():.1f}  med={tempos_unif.median():.1f}  '
      f'σ={tempos_unif.std():.1f}  skew={tempos_unif.skew():.2f}')

#### Minha análise — Exercício 8.2

A substituição da distribuição log-normal por uma uniforme expôs o trade-off fundamental da avaliação de dados sintéticos: não existe base "melhor" em absoluto, apenas bases mais adequadas a um critério específico.

Concretamente:

1. **Fidelidade estatística piorou drasticamente.** O teste KS rejeita a hipótese de log-normalidade na base uniforme — a assimetria natural do tempo de leitura (skewness positiva, com cauda direita representando sessões de leitura profunda) desaparece completamente.

2. **A coerência interna foi parcialmente preservada.** A diferença de mediana entre quem clicou e quem não clicou em comunicados aumentou — mas isso é um artefato da dispersão maior da uniforme, não uma melhora real na separação dos grupos.

3. **O realismo sofreu.** A distribuição uniforme é visualmente "plana demais" — qualquer analista com experiência em dados de navegação reconheceria imediatamente que não se trata de dados reais.

O ponto aqui é que avaliar qualidade exige considerar todos os eixos simultaneamente. Otimizar uma dimensão isoladamente pode dar uma falsa sensação de melhoria enquanto compromete o conjunto.
