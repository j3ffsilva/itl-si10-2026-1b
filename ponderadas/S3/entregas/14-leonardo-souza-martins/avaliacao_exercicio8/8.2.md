```text
# Exercício 8.2 — Enganar o modelo
# Modificação: tempo_leitura_segundos deixa de seguir uma log-normal
# e passa a ser gerado por uma distribuição uniforme.

def gerar_tempos_uniforme(objetivo, usou_busca, faixa_horaria):
    """
    Versão modificada da função gerar_tempos.

    Mantemos tempo_busca com distribuição log-normal.
    Alteramos apenas tempo_leitura para distribuição uniforme.

    A média da uniforme é calibrada para ficar próxima da média esperada
    da log-normal original por objetivo de sessão.
    """

    params = {
        "buscar_sistema": (4.0, 0.55, 3.2, 0.60),
        "buscar_documento": (4.0, 0.55, 3.2, 0.60),
        "buscar_pessoa": (4.0, 0.55, 3.2, 0.60),
        "ler_comunicado": (2.9, 0.45, 4.3, 0.55),
        "acompanhar_noticias": (3.2, 0.50, 4.0, 0.60),
    }

    mu_b, sig_b, mu_l, sig_l = params.get(objetivo, (3.5, 0.50, 3.8, 0.55))

    # Mantém a busca como log-normal
    tempo_busca = np.random.lognormal(mu_b, sig_b)

    # Média esperada da log-normal original:
    # E[X] = exp(mu + sigma² / 2)
    media_leitura_lognormal = np.exp(mu_l + (sig_l ** 2) / 2)

    # Distribuição uniforme calibrada em torno da média esperada.
    # Isso preserva uma média parecida, mas remove a cauda direita da log-normal.
    lim_inf = max(10, media_leitura_lognormal * 0.35)
    lim_sup = media_leitura_lognormal * 1.65

    tempo_leitura = np.random.uniform(lim_inf, lim_sup)

    if not usou_busca:
        tempo_busca *= 0.45

    if faixa_horaria == "08-11":
        tempo_leitura *= 0.88
        tempo_busca *= 1.08

    ruido = np.random.gamma(shape=2.0, scale=8.0)

    tempo_total = tempo_busca + tempo_leitura + ruido

    return tempo_busca, tempo_leitura, tempo_total
```

```text
from scipy import stats

def teste_aderencia_lognormal(serie):
    """
    Testa se uma variável positiva se ajusta bem a uma distribuição log-normal.

    Interpretação:
    - KS menor indica melhor aderência.
    - p-valor muito baixo indica evidência contra a hipótese de aderência log-normal.
    """

    x = serie.dropna()
    x = x[x > 0].to_numpy()

    shape, loc, scale = stats.lognorm.fit(x, floc=0)

    ks_stat, p_valor = stats.kstest(
        x,
        "lognorm",
        args=(shape, loc, scale)
    )

    return {
        "n": len(x),
        "sigma_estimado": shape,
        "mediana_estimativa": scale,
        "ks_stat": ks_stat,
        "p_valor": p_valor,
        "assimetria": stats.skew(x),
    }


resultado_original = teste_aderencia_lognormal(df["tempo_leitura_segundos"])
resultado_uniforme = teste_aderencia_lognormal(df_uniforme["tempo_leitura_segundos"])

df_teste_aderencia = pd.DataFrame([
    {
        "base": "Original log-normal",
        **resultado_original,
    },
    {
        "base": "Modificada uniforme",
        **resultado_uniforme,
    },
])

display(
    df_teste_aderencia.assign(
        sigma_estimado=lambda d: d["sigma_estimado"].map(lambda x: f"{x:.3f}"),

        p_valor=lambda d: d["p_valor"].map(lambda x: f"{x:.4g}"),

    )
)

print("Interpretação:")
print("A base uniforme deve apresentar pior aderência à log-normal.")
print("Isso aparece principalmente por KS maior, p-valor menor e menor cauda direita.")
```

```text
def avaliar_coerencia_tempo_clique(base, nome_base):
    mediana_nao_clicou = base[base["clicou_comunicado"] == 0]["tempo_leitura_segundos"].median()
    mediana_clicou = base[base["clicou_comunicado"] == 1]["tempo_leitura_segundos"].median()
    diferenca = mediana_clicou - mediana_nao_clicou

    correlacao = base[["tempo_leitura_segundos", "clicou_comunicado"]].corr().iloc[0, 1]

    return {
        "base": nome_base,
        "mediana_nao_clicou": mediana_nao_clicou,
        "mediana_clicou": mediana_clicou,
        "diferenca_segundos": diferenca,
        "correlacao_tempo_clique": correlacao,
        "coerencia_preservada": mediana_clicou > mediana_nao_clicou,
    }


coerencia_original = avaliar_coerencia_tempo_clique(df, "Original log-normal")
```
