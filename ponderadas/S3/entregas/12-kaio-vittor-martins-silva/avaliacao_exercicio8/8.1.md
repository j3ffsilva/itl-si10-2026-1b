# ─── PASSO 4 | Visualização comparativa ─────────────────────────────────────
import plotly.graph_objects as go
from plotly.subplots import make_subplots

fig = make_subplots(
    rows=1, cols=2,
    subplot_titles=[
        'Taxa de clique em comunicado por perfil',
        'Distribuição de objetivos de sessão',
    ],
)

for label, df_plot, cor in [
    ('Original (Oper.=18%)', df_original, '#94a3b8'),
    ('Cenário (Oper.=30%)',  df_cenario,  '#1d4ed8'),
]:
    tx = (df_plot.groupby('perfil_usuario')['clicou_comunicado']
          .mean().reset_index()
          .rename(columns={'clicou_comunicado': 'taxa'})
          .sort_values('taxa'))
    fig.add_trace(
        go.Bar(x=tx['taxa'], y=tx['perfil_usuario'], orientation='h',
               name=label, marker_color=cor, opacity=0.80,
               text=tx['taxa'].apply(lambda x: f'{x:.0%}'), textposition='outside'),
        row=1, col=1,
    )

for label, df_plot, cor in [
    ('Original', df_original, '#94a3b8'),
    ('Cenário',  df_cenario,  '#1d4ed8'),
]:
    dist_obj = df_plot['objetivo_sessao'].value_counts(normalize=True).reset_index()
    dist_obj.columns = ['objetivo', 'proporcao']
    fig.add_trace(
        go.Bar(x=dist_obj['objetivo'], y=dist_obj['proporcao'],
               name=label, marker_color=cor, opacity=0.80, showlegend=False,

    height=450,

fig.update_xaxes(tickformat='.0%', row=1, col=1)
