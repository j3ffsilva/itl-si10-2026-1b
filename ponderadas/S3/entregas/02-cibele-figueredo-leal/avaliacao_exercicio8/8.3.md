def bootstrap_diferenca_taxa(df_original, df_cenario, coluna, n_boot=1000, seed=42):
    np.random.seed(seed)

    diferencas = []

    for _ in range(n_boot):
        amostra_original = df_original[coluna].sample(
            n=len(df_original),
            replace=True
        )

        amostra_cenario = df_cenario[coluna].sample(
            n=len(df_cenario),
            replace=True
        )

        diferenca = amostra_cenario.mean() - amostra_original.mean()
        diferencas.append(diferenca)

    return np.array(diferencas)


diferencas_boot = bootstrap_diferenca_taxa(
    df_original=df,
    df_cenario=df_cenario_operacional,
    coluna='clicou_comunicado',
